1. **Create `tests/e2e/catalog.spec.ts`**:
   ```typescript
   import { test, expect } from '@playwright/test'

   test.describe('Public Catalog', () => {
     test('displays catalog page with product grid', async ({ page }) => {
       await page.goto('/es/catalogo')
       // Verify page title
       await expect(page).toHaveTitle(/Catalogo/)
       // Verify grid container exists
       await expect(page.locator('[data-testid="product-grid"]')).toBeVisible()
     })

     test('filters update URL query params', async ({ page }) => {
       await page.goto('/es/catalogo')
       // Click a filter checkbox (material)
       const filterSection = page.locator('[data-testid="filter-material"]')
       if (await filterSection.isVisible()) {
         const firstCheckbox = filterSection.locator('input[type="checkbox"]').first()
         await firstCheckbox.check()
         // URL should contain material param
         await expect(page).toHaveURL(/material=/)
       }
     })

     test('pagination navigates between pages', async ({ page }) => {
       await page.goto('/es/catalogo')
       const pagination = page.locator('[data-testid="pagination"]')
       if (await pagination.isVisible()) {
         const page2Link = pagination.locator('a:has-text("2")')
         if (await page2Link.isVisible()) {
           await page2Link.click()
           await expect(page).toHaveURL(/pagina=2/)
         }
       }
     })

     test('empty state shows when no results', async ({ page }) => {
       // Use impossible filter combination
       await page.goto('/es/catalogo?precio_min=99999999')
       await expect(page.getByText('Sin resultados')).toBeVisible()
     })

     test('product card links to detail page', async ({ page }) => {
       await page.goto('/es/catalogo')
       const firstCard = page.locator('[data-testid="product-card"]').first()
       if (await firstCard.isVisible()) {
         await firstCard.click()
         await expect(page).toHaveURL(/\/es\/catalogo\//)
       }
     })
   })

   test.describe('Product Detail', () => {
     test('shows 404 for nonexistent product', async ({ page }) => {
       const response = await page.goto('/es/catalogo/nonexistent-product-slug-xyz')
       expect(response?.status()).toBe(404)
     })
   })

   test.describe('Artisan Profile', () => {
     test('shows 404 for nonexistent artisan', async ({ page }) => {
       const response = await page.goto('/es/artesanos/nonexistent-artisan-slug-xyz')
       expect(response?.status()).toBe(404)
     })
   })
   ```
