   ```typescript
   import type { ProductWithDetails, ArtisanProfile } from '@/types/catalog.types'

   export function generateProductSchema(product: ProductWithDetails) {
     const coverMedia = product.media.find(m => m.is_cover) ?? product.media[0]
     return {
       '@context': 'https://schema.org',
       '@type': 'Product',
       name: product.title,
       description: product.description ?? '',
       image: coverMedia?.url ?? '',
       offers: {
         '@type': 'Offer',
         price: product.base_price,
         priceCurrency: 'CLP',
         availability: product.status === 'sold'
           ? 'https://schema.org/SoldOut'
           : 'https://schema.org/InStock',
       },
       brand: { '@type': 'Brand', name: product.artisan.user.full_name },
     }
   }

   export function generateArtisanSchema(artisan: ArtisanProfile, siteUrl: string) {
     return {
       '@context': 'https://schema.org',
       '@type': 'Person',
       name: artisan.user.full_name,
       description: artisan.bio ?? '',
       image: artisan.photo_url ?? '',
       url: `${siteUrl}/es/artesanos/${artisan.slug}`,
     }
   }
   ```

   import type { Metadata } from 'next'
   import type { ProductWithDetails, ArtisanProfile } from '@/types/catalog.types'

   export function generateProductMetadata(product: ProductWithDetails): Metadata {
     const coverMedia = product.media.find(m => m.is_cover) ?? product.media[0]
     const description = product.description?.slice(0, 155) ?? `${product.title} por ${product.artisan.user.full_name}`
     return {
       title: `${product.title} por ${product.artisan.user.full_name} | Crisol`,
       description,
       openGraph: {
         title: `${product.title} por ${product.artisan.user.full_name} | Crisol`,
         description,
         images: coverMedia ? [{ url: coverMedia.url }] : [],
       },
     }
   }

   export function generateArtisanMetadata(artisan: ArtisanProfile): Metadata {
     const description = artisan.bio?.slice(0, 155) ?? `${artisan.user.full_name} - Artesano en Crisol`
     return {
       title: `${artisan.user.full_name} - Artesano | Crisol`,
       description,
       openGraph: {
         title: `${artisan.user.full_name} - Artesano | Crisol`,
         description,
         images: artisan.photo_url ? [{ url: artisan.photo_url }] : [],
       },
     }
   }
