**Auth callback route** (`src/app/[locale]/auth/callback/route.ts`):
Replace placeholder with proper code exchange:
```typescript
import { createClient } from '@/lib/supabase/server'
import { NextResponse } from 'next/server'

export async function GET(request: Request) {
  const { searchParams, origin } = new URL(request.url)
  const code = searchParams.get('code')
  const next = searchParams.get('next') ?? '/es'

  if (code) {
    const supabase = await createClient()
    const { error } = await supabase.auth.exchangeCodeForSession(code)
    if (!error) {
      return NextResponse.redirect(`${origin}${next}`)
    }
  }

  return NextResponse.redirect(`${origin}/es/auth/login?error=auth_callback_error`)
}
```

export default async function ArtesanoLayout({ children }: { children: React.ReactNode }) {
  const supabase = await createClient()
  const { data: { user } } = await supabase.auth.getUser()

  if (!user) redirect('/es/auth/login')

  const { data: profile } = await supabase
    .from('user')
    .select('role')
    .eq('id', user.id)
    .single()

  if (profile?.role !== 'artisan' && profile?.role !== 'admin') {
    redirect('/es')
  }

  return <div className="min-h-screen">{children}</div>
}

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Browser -> Auth pages | User input (email, password) sent to Supabase Auth |
| Auth callback -> Session | Code exchange sets httpOnly cookie |
| Layout guards -> DB | Server component queries user table for role check |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-01 | Spoofing | Auth callback | mitigate | exchangeCodeForSession validates the code with Supabase; invalid codes redirect to login with error |
| T-03-02 | Elevation of Privilege | Layout guards | mitigate | Guards query 'user' table for role (not user-editable metadata); redirect on mismatch |
| T-03-03 | Information Disclosure | Auth error messages | mitigate | Generic error messages; no distinction between "user not found" and "wrong password" (Supabase default) |
| T-03-04 | Tampering | Session cookies | mitigate | @supabase/ssr handles httpOnly secure cookies; middleware refreshes on every request |
| T-03-05 | Denial of Service | Registration | accept | Supabase Auth has built-in rate limiting on signUp; no custom rate limiting needed at this layer |
</threat_model>
