  const TIMEOUT_MS = 5000
  export async function quote(courier, params, deps={quoteChilexpress, quoteStarken}) {
    if (params.destination.countryCode !== 'CL') throw new Error('CL_ONLY')
    const fn = courier === 'chilexpress' ? deps.quoteChilexpress : deps.quoteStarken
    try {
      return await Promise.race([fn(params), timeoutAfter(TIMEOUT_MS)])
    } catch (e) {
      console.error('[couriers]', courier, e)
      return { ...flatRateFallback(params), warning: `${courier}_failed` }
    }
  }
  export async function quoteForCart({destination, groups, preferredCourier='chilexpress'}) {
    return { quotes: await Promise.all(groups.map(async g => ({ artisanId: g.artisanId, quote: await quote(preferredCourier, {origin:g.origin, destination, package:g.package}) }))) }
  }
  ```

const schema = z.object({
  destination: z.object({ region: z.string(), comuna: z.string(), postalCode: z.string().optional(), countryCode: z.literal('CL') }),
  groups: z.array(z.object({
    artisanId: z.string().uuid(),
    origin: z.object({ region: z.string(), comuna: z.string(), postalCode: z.string().optional() }),
    package: z.object({ weightKg: z.number().positive(), lengthCm: z.number().positive(), widthCm: z.number().positive(), heightCm: z.number().positive() })
  })).min(1).max(10),
  preferredCourier: z.enum(['chilexpress','starken']).optional()
})
export async function POST(req: Request) {
  try {
    const input = schema.parse(await req.json())
    const result = await quoteForCart(input)
    return NextResponse.json(result)
  } catch (e: any) {
    if (e.message === 'CL_ONLY') return NextResponse.json({ error: 'Solo Chile' }, { status: 400 })
    if (e.name === 'ZodError') return NextResponse.json({ error: 'invalid_input', details: e.errors }, { status: 400 })
    return NextResponse.json({ error: 'server_error' }, { status: 500 })
  }
}
```
