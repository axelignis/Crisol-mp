```sql
CREATE OR REPLACE FUNCTION decrement_stock_atomic(p_items jsonb)
RETURNS TABLE(variant_id uuid, available int, requested int, ok boolean)
LANGUAGE plpgsql AS $$
DECLARE r record;
BEGIN
  FOR r IN SELECT (e->>'variantId')::uuid AS vid, (e->>'qty')::int AS qty
           FROM jsonb_array_elements(p_items) e LOOP
    PERFORM 1 FROM product_variant WHERE id = r.vid FOR UPDATE;
    IF (SELECT stock FROM product_variant WHERE id = r.vid) < r.qty THEN
      RETURN QUERY SELECT r.vid,
        (SELECT stock FROM product_variant WHERE id = r.vid),
        r.qty, false;
    ELSE
      UPDATE product_variant SET stock = stock - r.qty WHERE id = r.vid;
      RETURN QUERY SELECT r.vid,
        (SELECT stock FROM product_variant WHERE id = r.vid),
        r.qty, true;
    END IF;
  END LOOP;
END $$;
```
