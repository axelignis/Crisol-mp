**guest-token.ts**: HMAC-SHA256 sobre `${orderId}:${email}` con secret `GUEST_TOKEN_SECRET` (env nuevo) o reusar `REVALIDATE_SECRET` con namespace prefix. Recomendación: nuevo `GUEST_TOKEN_SECRET` (research D-V6). 
```ts
export function signGuestToken(orderId: string, email: string): string {
  const payload = `${orderId}:${email.toLowerCase()}`
  const sig = crypto.createHmac('sha256', process.env.GUEST_TOKEN_SECRET!).update(payload).digest('hex')
  return Buffer.from(`${payload}.${sig}`).toString('base64url')
}
export function verifyGuestToken(token: string): { orderId: string; email: string } | null {
  try {
    const decoded = Buffer.from(token, 'base64url').toString('utf8')
    const lastDot = decoded.lastIndexOf('.')
    const payload = decoded.slice(0, lastDot), sig = decoded.slice(lastDot + 1)
    const expected = crypto.createHmac('sha256', process.env.GUEST_TOKEN_SECRET!).update(payload).digest('hex')
    if (!crypto.timingSafeEqual(Buffer.from(sig), Buffer.from(expected))) return null
    const [orderId, email] = payload.split(':')
    return { orderId, email }
  } catch { return null }
}
```

export async function sendOrderConfirmedEmail(orderId: string): Promise<void> {
  const supabase = createServiceRoleClient()
  const { data: order } = await supabase.from('order').select('*, order_item(*), shipping_address(*), shipment(*)').eq('id', orderId).single()
  if (!order) return
  const recipient = order.buyer_id ? (await getUserEmail(order.buyer_id)) : order.guest_email
  if (!recipient) return
  const token = !order.buyer_id ? signGuestToken(orderId, recipient) : undefined
  const resend = new Resend(process.env.RESEND_API_KEY!)
  const result = await resend.emails.send({
    from: process.env.RESEND_FROM_EMAIL!,
    to: recipient,
    subject: `Confirmación de pedido #${orderId.slice(0,8)}`,
    react: OrderConfirmedEmail({ order, magicLinkToken: token })
  })
  if (result.error) console.error('[email] order-confirmed failed', result.error)
}
```

export default async function OrderPage({ params, searchParams }: { params: { id: string; locale: string }, searchParams: { token?: string } }) {
  const supabase = createClient()
  const { data: { user } } = await supabase.auth.getUser()
  let authorized = false
  if (user) {
    // Buyer-owned: RLS allows; query and rely on Phase 1 policies
    const { data } = await supabase.from('order').select('id, buyer_id').eq('id', params.id).single()
    authorized = !!data && data.buyer_id === user.id
  } else if (searchParams.token) {
    const verified = verifyGuestToken(searchParams.token)
    if (verified?.orderId === params.id) {
      authorized = true
    }
  }
  if (!authorized) notFound()

  // Use service role (or RLS-allowed) to load full detail
  const admin = createServiceRoleClient()
  const { data: order } = await admin.from('order').select('*, order_item(*), shipping_address(*), shipment(*)').eq('id', params.id).single()
  if (!order) notFound()
  return <OrderDetailView order={order} />
}
```
