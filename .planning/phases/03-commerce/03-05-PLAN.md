```ts
export async function createOrderFromPayment({event, supabase}): Promise<{orderId: string}> {
  const pi = event.data.object  // PaymentIntent
  const snapshotId = pi.metadata.cart_snapshot_id
  if (!snapshotId) throw new Error('NO_SNAPSHOT_ID')
  // 1. Load snapshot
  const { data: snap } = await supabase.from('cart_snapshot').select('*').eq('id', snapshotId).single()
  if (!snap) throw new Error('SNAPSHOT_NOT_FOUND')
  // 2. Decrement stock atomically (RPC)
  const { data: stockResult } = await supabase.rpc('decrement_stock_atomic', { p_items: snap.payload.items })
  const insufficient = stockResult.filter(r => !r.ok)
  if (insufficient.length > 0) throw new StockInsufficientError(insufficient)
  // 3. Insert order (status=pending_payment initially per migration 005 default)
  const { data: order } = await supabase.from('order').insert({
    buyer_id: snap.buyer_id, guest_email: snap.buyer_id ? null : snap.email,
    subtotal: snap.totals.subtotal, shipping_cost: snap.totals.shippingTotal,
    discount_amount: snap.totals.discount, commission_amount: snap.totals.commission,
    commission_pct_snapshot: snap.totals.commissionPct, total: snap.totals.total,
    coupon_id: snap.totals.couponId ?? null
  }).select().single()
  // 4. Insert order_items (snapshot fields)
  await supabase.from('order_item').insert(snap.totals.perArtisan.flatMap(...))
  // 5. Insert shipping_address
  // 6. Insert shipment (one per artisan from snap.payload.shipments)
  // 7. Insert payment (method='stripe', stripe_payment_intent_id=pi.id, status='paid', stripe_transfer_id=null per D-SPLIT)
  // 8. Insert artisan_payout (one per artisan, status='pending', stripe_payout_id=null)
  // 9. Coupon: ya reservado en payment-intent endpoint (Plan 04 reserveCoupon). Aquí SOLO asociar coupon_id a la order (ya hecho en step 3 vía snap.totals.couponId). NO incrementar uses_count (evita doble incremento).
  // 10. Call transition_order_status(order.id, 'paid', null/system)
  return { orderId: order.id }
}
```

```ts
export async function POST(req: Request) {
  const body = await req.text()  // RAW — never req.json() before constructEvent
  const sig = req.headers.get('stripe-signature') ?? ''
  let event
  try {
    event = stripe.webhooks.constructEvent(body, sig, process.env.STRIPE_WEBHOOK_SECRET!)
  } catch (e: any) {
    console.error('[stripe-webhook] signature failed', e.message)
    return NextResponse.json({ error: 'invalid_signature' }, { status: 400 })
  }
  const supabase = createServiceRoleClient()
  // Idempotency PK insert (verify col name in migration 007 — assume `event_id` or `id`)
  const { error: idemErr } = await supabase.from('webhook_idempotency').insert({ event_id: event.id, source: 'stripe', event_type: event.type })
  if (idemErr) {
    if (idemErr.code === '23505') return NextResponse.json({ received: true, duplicate: true })
    console.error('[stripe-webhook] idem insert failed', idemErr)
    return NextResponse.json({ error: 'idempotency_store_failed' }, { status: 500 })
  }
  switch (event.type) {
    case 'payment_intent.succeeded':
      try {
        const { orderId } = await createOrderFromPayment({ event, supabase })
        // Email send: await sendOrderConfirmedEmail(orderId) — wired in Plan 06
        return NextResponse.json({ received: true, orderId })
      } catch (e: any) {
        console.error('[stripe-webhook] order creation failed', e)
        // CRITICAL: payment already succeeded; do NOT throw 5xx that triggers Stripe retry+duplicate processing
        // Instead: log + return 200 + flag for manual reconciliation (insert into webhook_idempotency.error_message)
        await supabase.from('webhook_idempotency').update({ error_message: String(e?.message ?? e) }).eq('event_id', event.id)
        return NextResponse.json({ received: true, error: 'order_creation_failed', details: e.message })
      }
    case 'payment_intent.payment_failed':
      console.warn('[stripe-webhook] payment failed', event.data.object.id)
      // Phase 3 minimum: log only. No order created. Phase 4 may add alerts.
      return NextResponse.json({ received: true })
    default:
      return NextResponse.json({ received: true, ignored: event.type })
  }
}
```
